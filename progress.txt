  Verification:
    docker compose up -d --build

    # Register the admin user for a workspace (first user becomes admin)
    curl -sS -X POST http://localhost:8000/api/v1/auth/register \
      -H 'Content-Type: application/json' \
      -d '{"workspace_name":"Demo Workspace","email":"admin@example.com","password":"AdminPass123!"}' \
      | jq -r .access_token > /tmp/admin.token

    # Create a user + trigger invite email
    curl -sS -X POST http://localhost:8000/api/v1/users/admin-create \
      -H 'Authorization: Bearer '"$(cat /tmp/admin.token)" \
      -H 'Content-Type: application/json' \
      -d '{"email":"john@example.com"}' | jq

    # Check MailHog:
    open http://localhost:8025
    # or API:
    curl -sS http://localhost:8025/api/v2/messages | jq '.items[0].Content.Headers.Subject[0]'
